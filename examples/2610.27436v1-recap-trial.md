在仓库根目录执行下列 Python 代码；它从本文件取上述四个 JSON 块和术语表，复制一份缓存论文到全新的临时目录，所有 CLI 调用显式使用该目录。没有 `fetch`、网络请求或对默认 `~/.paperread` 的访问；临时目录在退出时自动清理。程序先重放导读及旧句子，再检查术语表路径、保存文件并追加 `recap`；输出只报告开发检查，不表示真实交互成功。

```python
import hashlib
import json
import os
from pathlib import Path
import re
import shutil
import subprocess
import sys
import tempfile

repo = Path.cwd()
example = repo / "examples/2610.27436v1-recap-trial.md"
text = example.read_text(encoding="utf-8")
seeds = [json.loads(s) for s in re.findall(r"^```json\n(.*?)^\`\`\`$", text, re.M | re.S)]
assert len(seeds) == 4
overview, sentence, profile, recap = seeds
glossary = re.search(r"^```markdown\n(.*?)^\`\`\`$", text, re.M | re.S).group(1)
terms = [line.split("|")[1].strip() for line in glossary.splitlines()
         if line.startswith("| ")][2:]
assert len(terms) == 20 and terms == recap["terms"]
assert recap["covered"] == sentence["explained"]
assert "type" not in sentence and "segment" not in recap
assert sorted(c["status"] for c in profile["concepts"].values()) == [
    "explained", "explained", "unsure"
]

cache = repo / ".local/claude-smoke/papers/2610.27436v1"
cli = repo / "skills/paper-foundations/scripts/paperread.py"
assert (cache / "paper.json").is_file(), "需要已有本地缓存，本程序不联网获取"

with tempfile.TemporaryDirectory(prefix="paperread-recap-trial-") as temporary:
    trial = Path(temporary).resolve()
    paper = trial / "papers/2610.27436v1"
    paper.mkdir(parents=True)
    shutil.copyfile(cache / "paper.json", paper / "paper.json")
    if (cache / "source.html").is_file():
        shutil.copyfile(cache / "source.html", paper / "source.html")
    profile_path = trial / "profile.json"
    profile_path.write_text(json.dumps(profile, ensure_ascii=False), encoding="utf-8")
    env = dict(os.environ, PAPERREAD_HOME=str(trial), PYTHONDONTWRITEBYTECODE="1")

    def run(*args, record=None):
        result = subprocess.run(
            [sys.executable, str(cli), *args], cwd=repo, env=env,
            input=None if record is None else json.dumps(record, ensure_ascii=False),
            text=True, capture_output=True, check=True,
        )
        return json.loads(result.stdout)

    for record in (overview, sentence):
        run("log", "add", "2610.27436v1", "--json", "-", record=record)
    assert len(run("log", "show", "2610.27436v1", "--type", "sentence", "--last", "0")) == 1
    concepts = run("profile", "concepts", "--paper", "2610.27436v1")
    assert len(concepts) == 3
    artifact = run("glossary-path", "2610.27436v1")
    glossary_path = Path(artifact["path"])
    assert glossary_path.parent == paper and not artifact["exists"]
    glossary_path.write_text(glossary, encoding="utf-8")
    assert glossary_path.read_text(encoding="utf-8") == glossary
    before = hashlib.sha256(profile_path.read_bytes()).hexdigest()
    run("log", "add", "2610.27436v1", "--json", "-", record=recap)
    assert before == hashlib.sha256(profile_path.read_bytes()).hexdigest()
    records = run("log", "show", "2610.27436v1", "--type", "recap", "--last", "0")
    assert len(records) == 1 and records[0]["terms"] == terms
    assert run("glossary-path", "2610.27436v1")["exists"]
    print("PASS: 20 terms; 2 explained / 1 unsure / 0 known; recap leaves profile unchanged")
```
